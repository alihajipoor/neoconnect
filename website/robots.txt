# Deliberately permissive. This site has nothing worth hiding from a
# crawler and every reason to want the pages found: the whole SEO problem
# here is being discovered at all, not being over-indexed.

User-agent: *
Allow: /

# Neither of these is a page. inc/ holds templates that exit on direct
# request; data/ holds form state and the CSRF secret, in files that begin
# with <?php exit; ?> and so return nothing when requested. robots.txt is a
# request, not a control, and must never be the only thing stopping access
# to something.
Disallow: /inc/
Disallow: /data/

# The account area (/account/) is NOT disallowed, on purpose. It is a
# signed-in application, not content, and its page says so itself:
# <meta name="robots" content="noindex, follow">. A crawler that is not
# allowed to fetch the page never reads that line -- and with the Buy and
# Sign in links on every page pointing at it, the bare URL then gets
# indexed anyway, as a result with no description.

# Voucher short links resolve to a redirect and are handed out privately.
Disallow: /r/

# ?setlang= exists only to record a language choice and then redirect. It
# is not a distinct page and must not be indexed as one.
Disallow: /*?setlang=

# Sitemap. This is the one absolute URL on the site that inc/config.php
# cannot fill in for you -- update it if the domain ever changes.
#
# It points at the CONVENTIONAL path first. /sitemap.xml is what a crawler
# tries first; measured 2026-08-24 and 2026-10-19 it returned the home page
# as text/html, because the shared nginx sends any unknown path to
# /index.php. The router at the top of index.php now serves the sitemap
# there, with no server configuration. /sitemap.php is the file itself and
# is listed too, so a crawler never depends on the router alone.
Sitemap: https://neoxify.net/sitemap.xml
Sitemap: https://neoxify.net/sitemap.php
